    <script>
        // Clean Config Object
        const firebaseConfig = {
            apiKey: "AIzaSyBH1SZASAlS3GeCp93lQaNNei8q5m0d2bQ",
            authDomain: "angkor-math-2026.firebaseapp.com",
            projectId: "angkor-math-2026"
        };

        if (!firebase.apps.length) {
            firebase.initializeApp(firebaseConfig);
        }
        const db = firebase.firestore();

        // CLOUDINARY CONFIGURATION (Replace with your actual keys)
        const CLOUDINARY_CLOUD_NAME = "haxrpbly";
        const CLOUDINARY_UPLOAD_PRESET = "angkor_upload";

        let currentUser = null;
        let pendingDeleteId = null;
        let splashTimer = null;

        // Auto-dismiss Splash Screen after 3 seconds
        window.addEventListener('DOMContentLoaded', () => {
            splashTimer = setTimeout(dismissSplash, 3000);
        });

        function dismissSplash() {
            if (splashTimer) clearTimeout(splashTimer);
            const splash = document.getElementById('splash-screen');
            if (splash && !splash.classList.contains('hidden')) {
                splash.classList.add('opacity-0');
                setTimeout(() => {
                    splash.classList.add('hidden');
                    checkLocalSession();
                }, 500);
            }
        }

        function checkLocalSession() {
            const savedUser = localStorage.getItem('angkor_user');
            if (savedUser) {
                currentUser = JSON.parse(savedUser);
                showApp();
            } else {
                showAuth();
            }
        }

        async function handleAuth(type) {
            const usernameInput = document.getElementById('auth-username').value.trim();
            const passwordInput = document.getElementById('auth-password').value;
            const errorDiv = document.getElementById('auth-error');

            errorDiv.classList.add('hidden');
            if (!usernameInput || !passwordInput) {
                showAuthError("Please fill in both fields.");
                return;
            }

            showActionLoader("Authenticating...");

            try {
                const userRef = db.collection('users').doc(usernameInput.toLowerCase());
                const doc = await userRef.get();

                if (type === 'signup') {
                    if (doc.exists) {
                        hideActionLoader();
                        showAuthError("Username already exists. Choose a different one.");
                        return;
                    }
                    const newUser = { username: usernameInput, password: passwordInput, createdAt: new Date() };
                    await userRef.set(newUser);
                    currentUser = { username: usernameInput };
                } else if (type === 'login') {
                    if (!doc.exists || doc.data().password !== passwordInput) {
                        hideActionLoader();
                        showAuthError("Invalid username or password.");
                        return;
                    }
                    currentUser = { username: doc.data().username };
                }

                localStorage.setItem('angkor_user', JSON.stringify(currentUser));
                hideActionLoader();
                showApp();
            } catch (err) {
                hideActionLoader();
                showAuthError("Connection error. Ensure Firestore rules are set to public read/write.");
            }
        }

        function showAuthError(msg) {
            const errorDiv = document.getElementById('auth-error');
            errorDiv.textContent = msg;
            errorDiv.classList.remove('hidden');
        }

        function logout() {
            localStorage.removeItem('angkor_user');
            currentUser = null;
            document.getElementById('app-section').classList.add('hidden');
            showAuth();
        }

        function showAuth() {
            document.getElementById('auth-section').classList.remove('hidden');
        }

        function showApp() {
            document.getElementById('auth-section').classList.add('hidden');
            document.getElementById('app-section').classList.remove('hidden');
            document.getElementById('user-display').textContent = currentUser.username;
            listenToFiles();
        }

        function listenToFiles() {
            db.collection('files').orderBy('createdAt', 'desc').onSnapshot(snapshot => {
                const questionsContainer = document.getElementById('list-questions');
                const answersContainer = document.getElementById('list-answers');
                
                questionsContainer.innerHTML = '';
                answersContainer.innerHTML = '';

                let qCount = 0;
                let aCount = 0;

                snapshot.forEach(doc => {
                    const item = doc.data();
                    const id = doc.id;
                    const element = createFileCard(id, item);

                    if (item.category === 'Questions') {
                        questionsContainer.appendChild(element);
                        qCount++;
                    } else {
                        answersContainer.appendChild(element);
                        aCount++;
                    }
                });

                document.getElementById('count-questions').textContent = `${qCount} files`;
                document.getElementById('count-answers').textContent = `${aCount} files`;
            });
        }

        async function submitUpload(e) {
            e.preventDefault();
            const title = document.getElementById('file-title').value;
            const category = document.getElementById('file-category').value;
            const description = document.getElementById('file-desc').value;
            const fileInput = document.getElementById('file-input').files[0];

            if (!fileInput) return;

            if (CLOUDINARY_CLOUD_NAME === "YOUR_CLOUDINARY_CLOUD_NAME") {
                alert("Please add your Cloudinary Cloud Name and Upload Preset in the script section.");
                return;
            }

            closeUploadModal();
            showActionLoader("Uploading file to Cloud...");

            try {
                // Upload directly to Cloudinary CDN
                const formData = new FormData();
                formData.append("file", fileInput);
                formData.append("upload_preset", CLOUDINARY_UPLOAD_PRESET);

                const cloudRes = await fetch(`https://api.cloudinary.com/v1_1/${CLOUDINARY_CLOUD_NAME}/auto/upload`, {
                    method: "POST",
                    body: formData
                });

                if (!cloudRes.ok) {
                    throw new Error("Failed to upload to Cloudinary CDN.");
                }

                const cloudData = await cloudRes.json();

                showActionLoader("Saving metadata...");

                // Store CDN link in Firestore
                await db.collection('files').add({
                    title,
                    category,
                    description,
                    fileData: cloudData.secure_url,
                    fileName: fileInput.name,
                    uploader: currentUser.username,
                    createdAt: firebase.firestore.FieldValue.serverTimestamp()
                });

                hideActionLoader();
                showActionLoader("Success!");
                setTimeout(hideActionLoader, 1200);
                document.getElementById('upload-form').reset();
            } catch (err) {
                hideActionLoader();
                alert("Upload failed. Check your Cloudinary configuration or network connection.");
            }
        }

        function createFileCard(id, item) {
            const card = document.createElement('div');
            card.className = "border border-slate-200 rounded-md p-4 bg-slate-50 flex flex-col justify-between hover:border-slate-300 transition";
            card.innerHTML = `
                <div>
                    <div class="flex justify-between items-start mb-1">
                        <h4 class="font-bold text-slate-800 text-sm">${escapeHtml(item.title)}</h4>
                        <span class="text-xs text-slate-400 font-mono">${escapeHtml(item.uploader)}</span>
                    </div>
                    <p class="text-xs text-slate-600 mb-3 line-clamp-2">${escapeHtml(item.description)}</p>
                </div>
                <div class="flex justify-between items-center pt-2 border-t border-slate-200 text-xs">
                    <a href="${item.fileData}" download="${item.fileName}" target="_blank" class="text-blue-900 font-semibold hover:underline">Download / View</a>
                    <button onclick="openDeleteModal('${id}')" class="text-red-600 hover:text-red-800 font-medium">Delete</button>
                </div>
            `;
            return card;
        }

        function openUploadModal() { document.getElementById('upload-modal').classList.remove('hidden'); }
        function closeUploadModal() { document.getElementById('upload-modal').classList.add('hidden'); }

        function openDeleteModal(docId) {
            pendingDeleteId = docId;
            document.getElementById('delete-modal').classList.remove('hidden');
        }
        function closeDeleteModal() {
            pendingDeleteId = null;
            document.getElementById('delete-modal').classList.add('hidden');
        }

        document.getElementById('confirm-delete-btn').addEventListener('click', async () => {
            if (!pendingDeleteId) return;
            const id = pendingDeleteId;
            closeDeleteModal();
            showActionLoader("Deleting...");

            try {
                await db.collection('files').doc(id).delete();
                hideActionLoader();
                showActionLoader("Deleted Successfully!");
                setTimeout(hideActionLoader, 1000);
            } catch (err) {
                hideActionLoader();
                alert("Error deleting resource.");
            }
        });

        function showActionLoader(text) {
            document.getElementById('action-loader-text').textContent = text;
            document.getElementById('action-loader').classList.remove('hidden');
        }
        function hideActionLoader() {
            document.getElementById('action-loader').classList.add('hidden');
        }
        function escapeHtml(str) {
            return str.replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' }[m]));
        }
